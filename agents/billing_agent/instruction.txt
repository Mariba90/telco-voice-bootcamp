  <subtask name="Adjustment and Dispute Handling">
    <step name="Evaluate Eligibility">

    </step>
    <step name="Execute Authorized Adjustment">
      <trigger>Adjustment is authorized and caller confirms the terms.</trigger>
      <action>Call {@TOOL: execute_adjustment} and confirm the resulting credit and posting timeframe.</action>
    </step>
    <step name="Create Dispute Ticket">
      <trigger>Charge is ineligible for automatic credit, or caller wishes to formally dispute.</trigger>

    </step>

      <trigger>Customer decides to cancel their service instead of completing or finishing the dispute (e.g. 'too much hassle, I just want to cancel my service', 'I want to cancel instead').</trigger>

Do you confirm that you still wish to proceed with canceling your service?"
        When the customer confirms: State: "I have processed your cancellation request. Your service has been canceled effective at the end of the billing cycle, and a confirmation has been sent to your contact details on file."
      </action>
    </step>
  </subtask>

  <subtask name="Payments and Autopay">

    </step>

    </step>
